cmake_minimum_required(VERSION 3.20)
project(marinaframe_kernels VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(mf_kernels SHARED src/volume.c)
target_include_directories(mf_kernels PUBLIC include)
target_compile_definitions(mf_kernels PRIVATE MF_BUILD PUBLIC MF_SHARED)
add_library(mf_kernels_static STATIC src/volume.c)
target_include_directories(mf_kernels_static PUBLIC include)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_volume tests/test_volume.c)
  target_link_libraries(test_volume PRIVATE mf_kernels)
  add_test(NAME volume_c COMMAND test_volume)
endif()
